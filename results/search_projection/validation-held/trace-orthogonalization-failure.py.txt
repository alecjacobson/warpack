  v=frame.f_locals;s=v.get('solver')

   data=dict(kind=v.get('kind'),width=v.get('width'),cached=v.get('cached'),constructor=type(s).__name__,status=s.status.numpy().tolist())
   for key in ['gx','lx','xscale','sx']:
    a=getattr(s,key).numpy();data[key]=dict(finite=bool(np.isfinite(a).all()),minimum=float(np.nanmin(a)),maximum=float(np.nanmax(a)))

   print('\nORTHOGONALIZATION_FAILURE '+json.dumps(data),flush=True)
 return trace
